{{/* Article navigation that skips section index pages as targets but keeps their child pages in reading order. */}} {{- $reversePagination := .Store.Get "reversePagination" | default false -}} {{- $root := .FirstSection | default .CurrentSection | default .Site.Home -}} {{- $scratch := newScratch -}} {{- $scratch.Set "entries" (slice) -}} {{- template "book-pager-collect" (dict "page" $root "scratch" $scratch) -}} {{- $entries := $scratch.Get "entries" -}} {{- $currentIndex := -1 -}} {{- range $index, $entry := $entries -}} {{- $page := index $entry "page" -}} {{- if eq $page.RelPermalink $.RelPermalink -}} {{- $currentIndex = $index -}} {{- end -}} {{- end -}} {{- $prev := false -}} {{- $next := false -}} {{- if ge $currentIndex 0 -}} {{- range seq (sub $currentIndex 1) -1 0 -}} {{- $entry := index $entries . -}} {{- if index $entry "target" -}} {{- $prev = index $entry "page" -}} {{- break -}} {{- end -}} {{- end -}} {{- range seq (add $currentIndex 1) (sub (len $entries) 1) -}} {{- $entry := index $entries . -}} {{- if index $entry "target" -}} {{- $next = index $entry "page" -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- if $reversePagination -}} {{- $swap := $prev -}} {{- $prev = $next -}} {{- $next = $swap -}} {{- end -}} {{- if eq .Params.prev false }} {{- if $reversePagination }}{{ $next = false }}{{ else }}{{ $prev = false }}{{ end -}} {{ else }} {{- with .Params.prev -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $next = . }}{{ else }}{{ $prev = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .Params.next false }} {{- if $reversePagination }}{{ $prev = false }}{{ else }}{{ $next = false }}{{ end -}} {{ else }} {{- with .Params.next -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $prev = . }}{{ else }}{{ $next = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- if $prev -}} {{- $linkTitle := partial "utils/title" $prev -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:ltr:rotate-180\"") -}} {{- $linkTitle -}} {{- end -}} {{- if $next -}} {{- $linkTitle := partial "utils/title" $next -}} {{- $linkTitle -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:rtl:-rotate-180\"") -}} {{- end -}}
{{- end -}} {{- define "book-pager-collect" -}} {{- $page := .page -}} {{- $scratch := .scratch -}} {{- if and $page (not $page.IsHome) (ne $page.Params.sidebar.exclude true) -}} {{- $scratch.Add "entries" (slice (dict "page" $page "target" (not $page.IsSection))) -}} {{- end -}} {{- with $page -}} {{- $children := where (union .Sections .RegularPages) "Params.sidebar.exclude" "!=" true -}} {{- range $children.ByWeight -}} {{- template "book-pager-collect" (dict "page" . "scratch" $scratch) -}} {{- end -}} {{- end -}} {{- end -}}
